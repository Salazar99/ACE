G: G((rst_n == 0) |=> (count == 0))    [FIFO1]
   G((rst_n == 0) |=> (empty == 1))    [FIFO2]
   G((rst_n == 0) |=> (full == 0))    [FIFO3]
   G((count == 0) |-> (empty == 1))    [FIFO4]
   G((empty == 1) |-> (count == 0))    [FIFO5]
   G((count == 8) |-> (full == 1))    [FIFO6]
   G((full == 1) |-> (count == 8))    [FIFO7]
   G((count >= 1) |-> (empty == 0))    [FIFO8]
   G((count <= 7) |-> (full == 0))    [FIFO9]
